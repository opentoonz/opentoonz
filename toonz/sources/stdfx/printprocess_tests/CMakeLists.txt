cmake_minimum_required(VERSION 3.18)
project(PrintProcessTests LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "Match the OpenToonz library build" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_executable(printprocess_test printprocess_test.cpp ../celnoise.cpp)
target_include_directories(printprocess_test PRIVATE ..)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(printprocess_test PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
foreach(CASE plates transparent tiles invalid)
    add_test(NAME printprocess_${CASE} COMMAND printprocess_test ${CASE})
    set_tests_properties(printprocess_${CASE} PROPERTIES
        ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1")
endforeach()

find_package(Qt5 5.14 REQUIRED COMPONENTS Core Gui OpenGL)
find_package(Boost CONFIG REQUIRED)
include(TestBigEndian)
test_big_endian(TOONZ_TEST_BIG_ENDIAN)
set(OPENTOONZ_LIBRARY_DIR "" CACHE PATH "Matching OpenToonz shared libraries")
add_executable(printprocess_native_test native_test.cpp ../celnoise.cpp ../stdfx.cpp ../gradients.cpp)
target_include_directories(printprocess_native_test PRIVATE .. ../../include)
target_link_libraries(printprocess_native_test PRIVATE Qt5::Core Qt5::Gui Qt5::OpenGL Boost::headers)
foreach(LIBRARY tnzcore tnzbase toonzlib)
    find_library(${LIBRARY}_PATH NAMES ${LIBRARY}
        PATHS "${OPENTOONZ_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(printprocess_native_test PRIVATE "${${LIBRARY}_PATH}")
endforeach()
if(TOONZ_TEST_BIG_ENDIAN)
    target_compile_definitions(printprocess_native_test PRIVATE TNZ_LITTLE_ENDIAN=0)
else()
    target_compile_definitions(printprocess_native_test PRIVATE TNZ_LITTLE_ENDIAN=1)
endif()
if(APPLE)
    target_compile_definitions(printprocess_native_test PRIVATE
        MACOSX __MACOS__ i386 x64 GL_SILENCE_DEPRECATION)
elseif(UNIX)
    target_compile_definitions(printprocess_native_test PRIVATE LINUX)
endif()
foreach(BPP 32 64 128)
    add_test(NAME printprocess_native_${BPP} COMMAND printprocess_native_test ${BPP})
    set_tests_properties(printprocess_native_${BPP} PROPERTIES TIMEOUT 20
        ENVIRONMENT "UBSAN_OPTIONS=halt_on_error=1;DYLD_LIBRARY_PATH=${OPENTOONZ_LIBRARY_DIR}")
endforeach()
