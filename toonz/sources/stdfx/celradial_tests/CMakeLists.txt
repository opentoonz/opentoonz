cmake_minimum_required(VERSION 3.18)
project(CelRadialTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt5 5.14 REQUIRED COMPONENTS Core Gui OpenGL)
get_filename_component(FX_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(TOONZ_SOURCE_DIR "${FX_SOURCE_DIR}/.." ABSOLUTE)
if(NOT OPENTOONZ_LIBRARY_DIR)
    message(FATAL_ERROR "Set OPENTOONZ_LIBRARY_DIR to matching OpenToonz shared libraries")
endif()
add_executable(celradial_test celradial_test.cpp
    "${FX_SOURCE_DIR}/stdfx.cpp" "${FX_SOURCE_DIR}/gradients.cpp" "${FX_SOURCE_DIR}/celmotion.cpp"
    "${FX_SOURCE_DIR}/celnoise.cpp")
target_include_directories(celradial_test PRIVATE
    "${FX_SOURCE_DIR}" "${TOONZ_SOURCE_DIR}/include")
include(TestBigEndian)
test_big_endian(BIG_ENDIAN)
if(BIG_ENDIAN)
    target_compile_definitions(celradial_test PRIVATE TNZ_LITTLE_ENDIAN=0)
else()
    target_compile_definitions(celradial_test PRIVATE TNZ_LITTLE_ENDIAN=1)
endif()
if(APPLE)
    target_compile_definitions(celradial_test PRIVATE MACOSX __MACOS__ i386 x64
        GL_SILENCE_DEPRECATION)
elseif(UNIX)
    target_compile_definitions(celradial_test PRIVATE LINUX)
endif()
foreach(LIBRARY tnzcore tnzbase tnzext toonzlib)
    find_library(${LIBRARY}_PATH NAMES ${LIBRARY}
        PATHS "${OPENTOONZ_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(celradial_test PRIVATE "${${LIBRARY}_PATH}")
endforeach()
target_link_libraries(celradial_test PRIVATE Qt5::Core Qt5::Gui Qt5::OpenGL)
enable_testing()
add_test(NAME celradial_raster COMMAND celradial_test)
if(APPLE)
    set_tests_properties(celradial_raster PROPERTIES
        ENVIRONMENT "DYLD_LIBRARY_PATH=${OPENTOONZ_LIBRARY_DIR}")
endif()
